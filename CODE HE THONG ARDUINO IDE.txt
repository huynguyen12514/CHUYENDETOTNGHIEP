// --- HÀM ĐỌC/GHI SỐ THỰC (FLOAT) VÀO EEPROM ---
void EEPROM_writeFloat(int address, float value) {
  byte* p = (byte*)(void*)&value;
  for (int i = 0; i < sizeof(value); i++) {
    EEPROM.write(address + i, p[i]);
  }
}

float EEPROM_readFloat(int address) {
  float value = 0.0;
  byte* p = (byte*)(void*)&value;
  for (int i = 0; i < sizeof(value); i++) {
    p[i] = EEPROM.read(address + i);
  }
  return value;
}

// Ngắt đo xung cảm biến lưu lượng
void IRAM_ATTR pulseCounter() {
  pulseCount++;
}

// Hàm tính trung bình động làm mượt số lưu lượng
float getFilteredFlow(float newSample) {
  flowSamples[sampleIndex] = newSample;
  sampleIndex = (sampleIndex + 1) % FILTER_SAMPLES;
  
  float sum = 0;
  for (int i = 0; i < FILTER_SAMPLES; i++) {
    sum += flowSamples[i];
  }
  return sum / FILTER_SAMPLES;
}

// Tải dữ liệu từ EEPROM hoặc Thư viện
void loadData(int id) {
  if (id < 1 || id >= MAX_PLANTS) id = 1;
  selectedPlantID = id;
  
  int s = EEPROM.read(100 + (id * 2));
  int st = EEPROM.read(101 + (id * 2));
  if (id <= 30) {
    activeStart = (s <= 5 || s > 95) ? library[id].startT : s;
    activeStop = (st <= 5 || st > 95) ? library[id].stopT : st;
  } else {
    activeStart = (s > 0) ? s : 40;
    activeStop = (st > 0) ? st : 70;
  }

  Blynk.virtualWrite(V9, activeStart); 
  Blynk.virtualWrite(V13, activeStop);
  Blynk.virtualWrite(V3, activeStart);
  Blynk.virtualWrite(V4, id);
  EEPROM.write(50, id);
  EEPROM.commit();

  // --- LOG CHO BƯỚC 1 & BƯỚC 2: ĐÃ ĐƯA % LÊN SÁT CẠNH CON SỐ ---
  Serial.print("Da lua chon cay trong so: ");
  Serial.println(selectedPlantID);
  Serial.print("Nguong bat bom = ");
  Serial.print(activeStart); 
  Serial.println("%");       
  Serial.print("Nguong tat bom = ");
  Serial.print(activeStop);  
  Serial.println("%");       
  Serial.println("EEPROM ghi nho thanh cong");
}

// --- HÀM GỬI DỮ LIỆU LÊN GOOGLE SHEET ---
void sendDataToGoogleSheet(int soil, float temp, float hum, float flow, float total, String pumpState) {
  if (WiFi.status() == WL_CONNECTED) {
    secureClient.setInsecure();
    String url = "https://script.google.com/macros/s/" + googleScriptID + "/exec?";
    url += "soil=" + String(soil);
    url += "&temp=" + String(temp);
    url += "&hum=" + String(hum);
    url += "&flow=" + String(flow);
    url += "&total=" + String(total);
    url += "&pump=" + pumpState;
    
    Serial.print("Dang gui du lieu len Google Sheet: ");
    Serial.println(url);
    if (http.begin(secureClient, url)) {
      http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);
      int httpCode = http.GET();
      if (httpCode > 0) {
        String payload = http.getString();
        Serial.print("Phan hoi tu Google: ");
        Serial.println(payload);
      } else {
        Serial.print("Loi ket noi HTTP: ");
        Serial.println(http.errorToString(httpCode).c_str());
      }
      http.end(); 
    }
  } else {
    Serial.println("Mat ket noi Wifi, khong the gui Google Sheet!");
  }
}

void setup() {
  Serial.begin(115200);
  
  Serial.println("Reading EEPROM...");
  
  EEPROM.begin(512); 
  dht.begin();
  setupLibrary(); 
  
  pinMode(pumpPin, OUTPUT);
  digitalWrite(pumpPin, HIGH);
  
  pinMode(flowSensorPin, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(flowSensorPin), pulseCounter, FALLING);
  for (int i = 0; i < FILTER_SAMPLES; i++) {
    flowSamples[i] = 0.0;
  }

  totalLiters = EEPROM_readFloat(150);
  if (isnan(totalLiters) || totalLiters < 0.0) {
    totalLiters = 0.0;
  }

  WiFiManager wm;
  wm.autoConnect("64130859_CDTN_IoT", "12345678");
  Blynk.config(BLYNK_AUTH_TOKEN);
  int lastPlant = EEPROM.read(50);
  
  Serial.print("Last Plant ID = ");
  Serial.println((lastPlant >= 1 && lastPlant < MAX_PLANTS) ? lastPlant : 1);
  Serial.println("Restoring configuration... ");
  
  loadData((lastPlant >= 1 && lastPlant < MAX_PLANTS) ? lastPlant : 1);
  
  Serial.println("Hien thi thanh cong");
  Blynk.virtualWrite(V2, 1);
}

void loop() {
  Blynk.run();
  unsigned long currentTime = millis();
  
  bool isPumpRunning = (digitalRead(pumpPin) == LOW);
  
  if ((currentTime - oldTime) > 500) {
    noInterrupts();
    long tempPulseCount = pulseCount;
    pulseCount = 0; 
    interrupts();   
    
    float rawFlowRate = 0.0;
    if (isPumpRunning && tempPulseCount > 0) {
      float pulsesPerSecond = (float)tempPulseCount * 2.0;
      if (pulsesPerSecond < 1200.0) {
        rawFlowRate = (pulsesPerSecond / calibrationFactor);
        totalLiters += (rawFlowRate / 120.0);
      }
    }

    flowRate = getFilteredFlow(rawFlowRate);
    if (!isPumpRunning) {
      flowRate = 0.0;
    }
    
    oldTime = currentTime;
  }

  if (isPumpRunning == false && lastPumpState == LOW) {
    EEPROM_writeFloat(150, totalLiters);
    EEPROM.commit();
    lastEEPROMWrite = millis();
  }
  
  if (isPumpRunning && (millis() - lastEEPROMWrite > 30000)) {
    EEPROM_writeFloat(150, totalLiters);
    EEPROM.commit();
    lastEEPROMWrite = millis();
  }
  lastPumpState = isPumpRunning;

  // --- CHU KỲ 3 GIÂY: CẬP NHẬT CẢM BIẾN & LOGIC BƠM ---
  if (millis() - lastSendBlynk > 3000) {
    int percent = constrain(map(analogRead(sensorPin), 1023, 350, 0, 100), 0, 100);
    float h = dht.readHumidity();
    float t = dht.readTemperature();

    Blynk.virtualWrite(V0, percent);      
    Blynk.virtualWrite(V1, flowRate);     
    Blynk.virtualWrite(V8, totalLiters);
    if (!isnan(h) && !isnan(t)) {
      Blynk.virtualWrite(V11, t);
      Blynk.virtualWrite(V12, h);
    }

    // --- LOGIC ĐIỀU KHIỂN BƠM & LOG BƯỚC 2 + BƯỚC 3 ĐỒNG BỘ ---
    if (systemEnabled) {
      int startLimit = activeStart + compensator;
      int stopLimit = activeStop + compensator;

      if (percent < startLimit) {
          // Báo cáo kiểm thử Bước 2: Khi độ ẩm xuống dưới ngưỡng Hstart thì kích hoạt bơm
          if (digitalRead(pumpPin) == HIGH) { 
            Serial.print("Do am dat = ");
            Serial.print(percent);
            Serial.println("%");
            Serial.println("Bom dang mo");
            Serial.println("Ly do: do am dat qua kho");
          }
          digitalWrite(pumpPin, LOW); // Bật bơm
      } 
      else if (percent > stopLimit) {
          digitalWrite(pumpPin, HIGH); // Tắt bơm
      }
    }  
    else {
      digitalWrite(pumpPin, HIGH);
    }
    
    lastSendBlynk = millis();
  }

  // --- CHU KỲ GỬI GOOGLE SHEET ---
  if (millis() - lastSendGoogleSheet > sendSheetInterval) {
    int percent = constrain(map(analogRead(sensorPin), 1023, 350, 0, 100), 0, 100);
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    String pState = (digitalRead(pumpPin) == LOW) ? "ON" : "OFF";
    if (isnan(h) || isnan(t)) {
      Serial.println("Loi: Cam bien DHT11 doc gia tri NaN! Tam thoi dung gui Google Sheet de tranh loi.");
    } else {
      sendDataToGoogleSheet(percent, t, h, flowRate, totalLiters, pState);
    }
    lastSendGoogleSheet = millis();
  }
}
