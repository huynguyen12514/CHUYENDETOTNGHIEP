#define BLYNK_TEMPLATE_NAME "CDTNHethongtuoinuocthongminhIoT"
#define BLYNK_AUTH_TOKEN "WAqDl20qBrr-YBDqiL09njva3s_T9pHG"

#include <ESP8266WiFi.h>
#include <BlynkSimpleEsp8266.h>
#include <EEPROM.h>
#include <WiFiManager.h> 
#include "DHT.h" 

// --- CẤU HÌNH PHẦN CỨNG ---
const int sensorPin = A0;      
const int pumpPin = D5;        
const int flowSensorPin = D2;  
#define DHTPIN D6              
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);

// --- BIẾN TOÀN CỤC ---
#define MAX_PLANTS 31 
volatile long pulseCount = 0;
float flowRate = 0.0;
float totalLiters = 0.0;
unsigned long oldTime = 0;
unsigned long lastSendBlynk = 0; 

int activeStart, activeStop, selectedPlantID = 1;
int inputStart = 0, inputStop = 0; 
int compensator = 0;    

// --- CẤU TRÚC THƯ VIỆN CÂY TRỒNG ---
struct PlantProfile {
  int startT; 
  int stopT;  
};

PlantProfile library[MAX_PLANTS];

// SỬA LỖI: Định nghĩa thông số riêng biệt cho từng cây để Slider thay đổi
void setupLibrary() {
  library[1]  = {45, 70}; library[2]  = {40, 65}; 
  library[3]  = {55, 80}; library[4]  = {50, 75};   
  library[5]  = {60, 85}; library[6]  = {42, 68};   
  library[7]  = {52, 78}; library[8]  = {48, 72}; 
  library[9]  = {35, 55}; library[10] = {58, 82};   
  library[11] = {62, 88}; library[12] = {38, 62}; 
  
  // Tạo dữ liệu khác nhau cho các cây từ 13 đến 30 để tránh trùng lặp
  for (int i = 13; i < MAX_PLANTS; i++) {
    library[i].startT = 30 + (i % 10) * 2; 
    library[i].stopT = library[i].startT + 20; 
  }
}

void IRAM_ATTR pulseCounter() { pulseCount++; }

// --- HÀM TẢI DỮ LIỆU TỪ EEPROM ---
void loadData(int id) {
  if (id < 1 || id >= MAX_PLANTS) id = 1;
  selectedPlantID = id;
  EEPROM.write(50, id); 
  
  int s = EEPROM.read(100 + (id * 2)); 
  activeStart = (s > 0 && s <= 100) ? s : library[id].startT;
  
  int st = EEPROM.read(101 + (id * 2));
  activeStop = (st > 0 && st <= 100) ? st : library[id].stopT;

  // Cập nhật giá trị lên Slider trên App ngay lập tức
  Blynk.virtualWrite(V9, activeStart);  
  Blynk.virtualWrite(V13, activeStop); 
  Blynk.virtualWrite(V3, activeStart);
  EEPROM.commit();
}

// --- XỬ LÝ ĐIỀU KHIỂN TỪ APP BLYNK ---
BLYNK_WRITE(V4) { loadData(param.asInt()); }
BLYNK_WRITE(V6) { compensator = param.asInt(); }
BLYNK_WRITE(V7) { if (param.asInt() == 1) { totalLiters = 0; Blynk.virtualWrite(V8, 0); } }
BLYNK_WRITE(V9) { inputStart = param.asInt(); }
BLYNK_WRITE(V13) { inputStop = param.asInt(); }

BLYNK_WRITE(V5) { // Lưu cấu hình tùy chỉnh cho cây hiện tại
  if (param.asInt() == 1 && inputStart > 0 && inputStop > inputStart) {
    activeStart = inputStart; 
    activeStop = inputStop;
    EEPROM.write(100 + (selectedPlantID * 2), activeStart);
    EEPROM.write(101 + (selectedPlantID * 2), activeStop);
    EEPROM.commit();
    Blynk.virtualWrite(V3, activeStart);
  }
}

void setup() {
  Serial.begin(115200);
  EEPROM.begin(512);
  dht.begin();
  setupLibrary(); 
  
  pinMode(pumpPin, OUTPUT);
  digitalWrite(pumpPin, HIGH); // Tắt bơm mặc định (Relay kích LOW)
  
  pinMode(flowSensorPin, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(flowSensorPin), pulseCounter, FALLING);

  WiFiManager wm;
  wm.autoConnect("CDTN_SmartIrrigation_IoT", "12345678");
  Blynk.config(BLYNK_AUTH_TOKEN);

  // Khôi phục cây trồng cuối cùng đã chọn
  int lastPlant = EEPROM.read(50);
  selectedPlantID = (lastPlant >= 1 && lastPlant < MAX_PLANTS) ? lastPlant : 1;
  loadData(selectedPlantID);
}

void loop() {
  Blynk.run();

  // 1. CÔNG THỨC ĐO LƯU LƯỢNG CHUẨN (Mỗi 1 giây)
  unsigned long currentTime = millis();
  if ((currentTime - oldTime) > 1000) {
    detachInterrupt(digitalPinToInterrupt(flowSensorPin));
    
    // Q = F / 7.5 (Lít/Phút)
    flowRate = (pulseCount / 7.5); 
    
    // Tổng lượng nước (Lít): 1 lít ~ 450 xung đối với YF-S201
    totalLiters += (pulseCount / 450.0); 
    
    pulseCount = 0;
    oldTime = currentTime;
    attachInterrupt(digitalPinToInterrupt(flowSensorPin), pulseCounter, FALLING);
  }

  // 2. GỬI DỮ LIỆU VÀ ĐIỀU KHIỂN BƠM (Mỗi 3 giây)
  if (millis() - lastSendBlynk > 3000) {
    int percent = constrain(map(analogRead(sensorPin), 1023, 350, 0, 100), 0, 100);
    float h = dht.readHumidity();
    float t = dht.readTemperature();

    Blynk.virtualWrite(V0, percent);      
    Blynk.virtualWrite(V1, flowRate);     
    Blynk.virtualWrite(V8, totalLiters);  
    if (!isnan(h) && !isnan(t)) {
      Blynk.virtualWrite(V11, t);
      Blynk.virtualWrite(V12, h);
    }

    // --- LOGIC ĐIỀU KHIỂN (KHÔNG DÙNG GIỜ) ---
    int startLimit = activeStart + compensator;
    int stopLimit = activeStop + compensator;

    if (percent < startLimit) {
        digitalWrite(pumpPin, LOW);  // Bật bơm (Kích mức thấp)
    } 
    else if (percent > stopLimit) {
        digitalWrite(pumpPin, HIGH); // Tắt bơm (Kích mức thấp)
    }
    
    lastSendBlynk = millis();
  }
}
